cmake_minimum_required(VERSION 3.16)
project(ThiefPuzzle LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(SFML 3 REQUIRED COMPONENTS Graphics)
set(GAME_SOURCES src/main.cpp src/Game.cpp src/Renderer.cpp src/Menu.cpp src/Solver.cpp)
add_executable(thief_puzzle ${GAME_SOURCES})
target_link_libraries(thief_puzzle PRIVATE SFML::Graphics)
add_executable(thief_puzzle_dev ${GAME_SOURCES})
target_link_libraries(thief_puzzle_dev PRIVATE SFML::Graphics)
target_compile_definitions(thief_puzzle_dev PRIVATE THIEF_PUZZLE_DEV=1)
